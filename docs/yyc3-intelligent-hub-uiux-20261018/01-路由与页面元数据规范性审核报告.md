### 3.1 元数据工厂 `lib/yyc3/metadata.ts`（新建·全站唯一组装出口）

```typescript
import type { Metadata } from "next"

import { BRAND, SITE_URL } from "@/lib/yyc3/brand"

/**
 * 页面元数据工厂 —— 全站唯一的页面级 metadata 组装出口。
 *
 * 规范（与根 layout 的 title.template 联动）：
 * - title 只传「页面名」，模板统一拼接为「页面名 - YanYuCloudCube集成中心系统」；
 *   禁止在页面侧自拼品牌后缀（会造成双后缀或分隔符漂移）。
 * - description 缺省回退品牌主描述；动态页必须传入内容驱动描述。
 * - path 提供时生成 canonical 与 openGraph.url，消除静态导出下的重复收录。
 */
export function createMetadata({
  title,
  description = BRAND.description,
  path,
}: {
  /** 页面名（不带品牌后缀），如「我的收藏」 */
  title: string
  /** 页面描述；动态页由数据驱动传入 */
  description?: string
  /** 规范路径（以 / 开头，含 trailingSlash），如 "/favorites/" */
  path?: string
}): Metadata {
  return {
    title,
    description,
    ...(path
      ? {
          alternates: { canonical: path },
          openGraph: {
            title: `${title} - ${BRAND.name}`,
            description,
            url: new URL(path, SITE_URL).toString(),
          },
        }
      : {}),
  }
}

/**
 * 动态路由元数据工厂 —— generateMetadata 专用。
 * 强制要求 description（动态页禁止回退全站描述，否则 SEO 语义被稀释）。
 */
export function createDynamicMetadata({
  title,
  description,
  path,
}: {
  title: string
  description: string
  path: string
}): Metadata {
  return createMetadata({ title, description, path })
}
```

### 3.4 动态页标准范式（generateStaticParams + generateMetadata 同源）

```typescript
import type { Metadata } from "next"
import ClientLoader from "./client-loader"

import { integrations } from "@/app/data/integrations"
import { createDynamicMetadata } from "@/lib/yyc3/metadata"

/**
 * 由数据源驱动生成全部应用详情页的静态路径。
 * 必须遍历完整 `integrations` 集合，禁止硬编码数量，否则静态导出下未预渲染的应用编号会直接 404。
 */
export function generateStaticParams() {
  return integrations.map((integration) => ({ id: integration.id }))
}

/** 动态元数据：「{应用名} - YanYuCloudCube集成中心系统」，描述取自集成数据源。 */
export async function generateMetadata({
  params,
}: {
  params: Promise<{ id: string }>
}): Promise<Metadata> {
  const { id } = await params
  const integration = integrations.find((i) => i.id === id)
  if (!integration) return {} // 理论不可达：静态参数已穷举；回退全站默认元数据

  return createDynamicMetadata({
    title: integration.name,
    description: integration.description,
    path: `/integrations/${id}/`,
  })
}

export default function IntegrationDetailPage() {
  return <ClientLoader />
}
```

中文分类段特殊处理（URL 编码对称）：`generateStaticParams` 产出 `encodeURIComponent(c)`，`generateMetadata` 内先 `decodeURIComponent` 再匹配；「全部分类」聚合项过滤不生成静态路由。

### 3.5 路由守卫 `components/auth/route-guard.tsx`（新建·三态约定）

```typescript
"use client"

import { usePathname, useRouter } from "next/navigation"
import { useEffect } from "react"

import { useAuth } from "@/app/context/auth-context"

/**
 * 路由权限守卫 —— 全站唯一的客户端鉴权出口。
 *
 * 背景：项目为 `output: "export"` 静态导出，无法使用 middleware.ts 做边缘守卫，
 * 因此守卫统一在受保护路由段的 layout 中以客户端组件形态挂载。
 *
 * 行为约定（三态，禁止业务页自行实现）：
 * 1. isLoading（会话恢复中）→ 渲染无障碍骨架（aria-busy），不泄露受保护内容；
 * 2. 未登录 → 重定向 `/login?redirect={当前路径}`，登录成功后原路回跳；
 * 3. 已登录 → 直接放行 children。
 */
export function RequireAuth({ children }: { children: React.ReactNode }) {
  const { isAuthenticated, isLoading } = useAuth()
  const router = useRouter()
  const pathname = usePathname()

  useEffect(() => {
    if (!isLoading && !isAuthenticated) {
      router.replace(`/login?redirect=${encodeURIComponent(pathname)}`)
    }
  }, [isAuthenticated, isLoading, pathname, router])

  if (isLoading) {
    return (
      <div
        className="flex min-h-screen items-center justify-center bg-background"
        role="status"
        aria-busy="true"
      >
        <div className="h-8 w-8 animate-spin rounded-full border-2 border-primary border-t-transparent" />
        <span className="sr-only">正在验证登录状态</span>
      </div>
    )
  }

  if (!isAuthenticated) {
    // 重定向进行中：渲染空白避免受保护内容闪现
    return null
  }

  return <>{children}</>
}
```

### 3.6 登录回跳承接 `app/login/page.tsx`（含开放重定向校验）

```typescript
"use client"

import { useSearchParams } from "next/navigation"
import { useRouter } from "next/navigation"
import { Suspense } from "react"

import { LoginForm } from "@/app/components/auth/login-form"
import { PageContainer } from "@/app/components/layout/page-container"

/** 登录成功后的回跳目标；仅接受站内路径，防开放重定向。 */
function safeRedirectTarget(raw: string | null): string {
  if (raw && raw.startsWith("/") && !raw.startsWith("//")) return raw
  return "/account/subscriptions"
}

/**
 * useSearchParams 必须位于 Suspense 边界内才能参与静态导出（否则构建期 CSR bailout）。
 */
function LoginSuccessHandler() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const redirectTo = safeRedirectTarget(searchParams.get("redirect"))

  return (
    <LoginForm
      onSuccess={() => router.replace(redirectTo)}
      onCancel={() => router.push("/")}
    />
  )
}

export default function LoginPage() {
  return (
    <PageContainer className="py-16">
      <Suspense fallback={null}>
        <LoginSuccessHandler />
      </Suspense>
    </PageContainer>
  )
}
```

## Layer 5 · 规范沉淀与后续建议

### 5.1 本次确立的路由规范（约束力：硬性）

1. **元数据唯一出口**：所有页面元数据必须经 `createMetadata` / `createDynamicMetadata` 工厂产出；页面侧只传「页面名」，禁止自拼品牌后缀。
2. **metadata 承载位置**：server layout 优先；client 页面（`"use client"`）禁止导出 metadata，元数据一律上移至其 layout。
3. **动态页三件套同源**：`generateStaticParams` 与 `generateMetadata` 必须消费同一数据源（`app/data/integrations.ts`），禁止硬编码副本；中文参数段 encodeURIComponent/decode 对称处理。
4. **守卫唯一出口**：受保护段（admin/account 及未来新增）必须在段 layout 挂载 `RequireAuth`，业务页禁止自行实现鉴权跳转；登录回跳固定 `?redirect=` 约定 + 站内路径白名单校验。
5. **激活态语义**：导航链接激活必须携带 `aria-current="page"`；判定统一走 `isActiveRoute` 前缀匹配。
6. **面包屑=真实路由链**：crumbs 必须与 URL 层级一一对应，禁止跨层聚合。
